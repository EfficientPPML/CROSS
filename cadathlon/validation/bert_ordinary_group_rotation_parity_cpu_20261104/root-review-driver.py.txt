"""Public source/result review only; does not import runtime or reread CT/key bodies."""
from pathlib import Path
from types import SimpleNamespace
import ast
import datetime
import hashlib
import json
import math
import os
import tarfile

ROOT = Path('/tmp/bert-ordinary-group-parity-ready-20261103')
OUT = ROOT / 'actual-v1'
DEST = Path(__file__).parent
sha = lambda data: hashlib.sha256(data).hexdigest()
read = lambda path: json.loads(path.read_bytes())
pins = {
    OUT/'report.json': 'dae30871ca548367b873f4fcf2c65e29867413c0286f83d124797f433befaea3',
    OUT/'worker/worker_report.json': 'f94a9be90a458a6b398c0245005f5504f3d42c82af26a771a2afffdad3961273',
    OUT/'sources.tar.gz': '56afba21d5e6f4c4805b202565b8f1bb6360b976de07369beb6dbf77bc371d59',
    OUT/'source_manifest.json': '5018516fa46b9200352a16c0ccf7de9e0de07ca7fbd4ee64c2b8bc216c14a22b',
    Path('/tmp/bert-ordinary-group-parity-terminal-20261104/owner-terminal.json'): 'd4f3d9d38bb7cb6942da284e56bc50c1bf79bbe9f1c6729b50310e2a18e3f8ff',
    Path('/tmp/check_ordinary_group_terminal_20261104.py'): '4425136d6bcf81a21e194a3699eb2cb94e95d47c1d2725a059b005423b433743',
}
for path, digest in pins.items():
    assert sha(path.read_bytes()) == digest, path
p = read(OUT/'report.json')
w = read(OUT/'worker/worker_report.json')
expected = read(OUT/'source_manifest.json')
assert len(expected) == 259
assert expected == p['source_hashes_start'] == p['source_hashes_end'] == p['execution_source_hashes_end'] == w['source_hashes_start'] == w['source_hashes_end']
for name, digest in expected.items():
    rel = Path(name)
    assert not rel.is_absolute() and '..' not in rel.parts
    for tree in (ROOT, OUT/'execution_source'):
        assert sha((tree/rel).read_bytes()) == digest, (tree, name)
with tarfile.open(OUT/'sources.tar.gz') as archive:
    members = archive.getmembers()
    assert len(members) == len({m.name for m in members}) == 259
    assert all(m.isfile() for m in members)
    assert {m.name: sha(archive.extractfile(m).read()) for m in members} == expected
for field in ('passed', 'execute', 'actual_rotation_parity', 'actual_execution_attempted', 'all_processes_reaped', 'external_workspaces_never_removed', 'operation_complete', 'terminal_scope_completed', 'source_files_unchanged', 'source_archive_verified', 'io_workspace_retained', 'cleanup_completed'):
    assert p[field] is True, field
for record in (p, w):
    assert not any(record.get(f) for f in ('error', 'cleanup_errors', 'forbidden_attempts'))
assert p['source_archive_sha256'] == pins[OUT/'sources.tar.gz']
proc = p['worker_process']
assert proc == read(OUT/'worker/process.json')
assert proc['result'] == read(OUT/'worker/stdout.json')
assert type(proc['returncode']) is int and proc['returncode'] == 0 and proc['timeout_s'] is None
assert proc['pid'] == w['pid'] == 1878109
assert not proc.get('error')
assert all(proc[f] is True for f in ('process_reaped', 'public_key_store_removed', 'owned_directory_handle_closed'))
assert proc['passed'] is False  # Preserved generic run_case initializer, not a selected proof gate.
assert not Path(proc['owned_public_key_directory']).exists()

# Execute only the selected collector's pure result gate, with constants and a
# file-hash function. No runtime imports, contexts, codec, model or CT access.
module = ast.parse((ROOT/'group_collect.py').read_text())
gate = next(node for node in module.body if isinstance(node, ast.FunctionDef) and node.name == 'validate_worker')
groups = (('head', tuple(63*i for i in range(12))), ('local_key', tuple(-16*i for i in range(16))), ('edges', (0,63,63,-16,0)))
scope = {'process': SimpleNamespace(sha256=lambda path: sha(Path(path).read_bytes())), 'group_protocol': SimpleNamespace(LEVELS=(49,25), GROUPS=groups)}
exec(compile(ast.Module(body=[gate], type_ignores=[]), str(ROOT/'group_collect.py'), 'exec'), scope)
prereq = p['prerequisite']
inputs = dict(p['request_readback'])
scope['validate_worker'](proc, w, expected, p, {'context_id': prereq['context_id']}, {'model_id': prereq['model_id']}, inputs, OUT/'worker/worker_report.json')
assert p['request_readback'] == w['request_readback']
assert p['arithmetic'] == w['arithmetic']
arith = w['arithmetic']
rows = arith['comparisons']
assert rows == w['completed_comparisons'] and len(rows) == 66
params = p['public_plan']['parameters']
assert params['degree'] == 131072 and params['num_slots'] == 65536 and params['dnum'] == 8 and params['composite_degree'] == 2
q, aux = params['q_towers'], params['p_towers']
assert len(q) == 100 and len(aux) == 14 and len(set(q+aux)) == 114
assert all(type(x) is int and x > 1 for x in q+aux)
assert (math.prod(q)*math.prod(aux)).bit_length() == 3403
assert all(math.prod(aux) >= math.prod(q[i:i+13]) for i in range(0,100,13))
for row in rows:
    towers = 2*(row['level']+1)
    value = row['left']
    assert row['right'] == value
    assert value['shape'] == [1,2,256,512,towers]
    assert value['moduli'] == q[:towers] and value['degree_layout'] == [256,512] and value['dtype'] == 'uint32'
    assert type(value['precision']) is int and value['precision'] == 32
    assert type(value['nsd']) is int and value['nsd'] == 1
    assert type(value['scale']) is float and value['scale'] == params['scaling_factor']
    assert value['scale_hex'] == value['scale'].hex()
    assert len(value['payload_sha256']) == 64 and all(c in '0123456789abcdef' for c in value['payload_sha256'])
    assert type(row['mismatched_words']) is int and row['mismatched_words'] == 0
assert sum(row['compared_words'] for row in rows) == 1314914304
reduction = arith['level_reduction']
assert all(reduction[f] is True for f in ('passed','payload_prefix_equal','ordered_q_prefix'))
assert reduction['input']['shape'] == [1,2,256,512,100] and reduction['output']['shape'] == [1,2,256,512,52]
assert reduction['input']['moduli'] == q and reduction['output']['moduli'] == q[:52]
assert reduction['input']['scale'] == reduction['output']['scale'] == params['scaling_factor']
origins = {}

    if name.startswith('loaded_local_modules'):
        for value in values.values():
            assert expected[value['path']] == value['sha256']
        origins[name] = len(values)
for pid in (1878080,1878082,1878109):

        os.killpg(pgid, 0)
    except ProcessLookupError:

borrowed = {}
for name, path, identity in (
    ('keys','/dev/shm/cross-full-bert-keys-143b3f5c1573427d841760a3c9d5d33d',(24,1563)),
    ('model','/tmp/cross-bert-public-model-b51668856dab42a3846ba1b7c7a71d53',(66305,2383041)),

result = {
    'schema': 'ordinary-group-parity-root-review-v1', 'passed': True,
    'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),

}
