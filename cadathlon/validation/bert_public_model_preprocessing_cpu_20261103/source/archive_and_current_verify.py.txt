        raw={x.name:tar.extractfile(x).read() for x in members};assert {n:hashlib.sha256(b).hexdigest() for n,b in raw.items()}==before

    tree=ast.parse(raw['checkout/stage/server_public_model.py']);pins=next(ast.literal_eval(n.value) for n in tree.body if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='TENSOR_SHA256' for t in n.targets))

        model_id=r['model_id'],context_id=r['context_id'],model_manifest_sha256=r['model_manifest_sha256'],public_tensor_count=196,public_matrix_count=74,
        worker_elapsed_s=w['elapsed_s'],collector_elapsed_s=r['elapsed_s'],cpu_affinity=r['cpu_affinity'],

        no_key_private_or_model_payloads_read_or_copied_for_archive=True,
