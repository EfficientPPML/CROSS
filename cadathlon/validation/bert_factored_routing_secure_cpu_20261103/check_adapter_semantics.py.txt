"""NumPy-only check of the exact encrypted-adapter operation schedule."""
from pathlib import Path
import hashlib,json,math
import numpy as np
from check_secure_routing import EncryptedMap
from score_layout import ScoreMap,SLOTS
HERE=Path(__file__).resolve().parent
paths=[Path(__file__),HERE/'check_secure_routing.py',HERE/'score_layout.py']
before={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in paths}
class Value:
    def __init__(self,slots,level):self.slots=slots;self.level=level;self._ckks_scale=1.;self._ckks_nsd=1
class Arithmetic:
    scale=1.
    def _level(self,value):return value.level
    def weighted_sum(self,values,weights):return Value(sum(v.slots*w for v,w in zip(values,weights)),min(v.level for v in values)-1)
    def add(self,left,right):
        assert left.level==right.level
        return Value(left.slots+right.slots,left.level)
    def rotate(self,value,shift):return Value(np.roll(value.slots,-shift),value.level)
mapping=ScoreMap(7,15);rng=np.random.default_rng(20261103)
values=rng.uniform(4.,8.,SLOTS)*np.where(np.arange(SLOTS)%2,-1.,1.)
values[mapping.source]=np.arange(192)/256.-.375
phases=[]
router=EncryptedMap(Arithmetic(),lambda label,value,counts:phases.append(dict(label=label,level=value.level)))
forward=router.apply(Value(values,18));backward=router.inverse(forward)
assert forward.level==9 and backward.level==0
np.testing.assert_array_equal(forward.slots,mapping.oracle(values))
np.testing.assert_array_equal(backward.slots,values*mapping.source_mask)
assert dict(router.counts)==dict(public_multiplications=34,actual_composite_rescales=34,public_multiplication_levels=18,rotations=18,additions=16)
after={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in paths}
report=dict(scope=__doc__,passed=before==after,source_hashes_before=before,source_hashes_after=after,source_files_unchanged=before==after,counts=dict(router.counts),phases=phases,full_slots_checked=SLOTS,forward_and_roundtrip_exact=True)
(HERE/'adapter_semantics.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({k:v for k,v in report.items() if k not in ('phases','source_hashes_before','source_hashes_after')},indent=2))
raise SystemExit(0 if report['passed'] else 1)
