def test_distinct_full_width_rows_and_real_diagonal_layout():
 from encrypted_packed_qwen import MatrixDiagonals
 rng=np.random.default_rng(58);rows=rng.normal(size=(2,768));slots=pack_rows(rows)
 assert not np.array_equal(rows[0],rows[1]);np.testing.assert_array_equal(unpack_rows(slots),rows)
 # Exercise the actual BSGS diagonal rotations/sums on the full768x768 width.
 w=rng.normal(scale=.02,size=(768,768));source=MatrixDiagonals(w,8192,4096);babies={i:np.roll(slots,-i) for i in {d%source.n1 for d in source.active}};actual=np.zeros(8192)
 for giant in sorted({d//source.n1*source.n1 for d in source.active}):
  indices=[d for d in source.active if d//source.n1*source.n1==giant];diags=source.materialize_diagonals(indices);inner=sum(np.roll(diags[d],giant)*babies[d%source.n1] for d in indices);actual+=np.roll(inner,-giant)
 expected=pack_rows(rows@w.T);np.testing.assert_allclose(actual,expected,atol=1e-12,rtol=1e-12)
 assert np.max(np.abs(actual.reshape(2,4096)[:,768:]))==0

@pytest.mark.parametrize('change',['context','request','role','input_digest','scale','nsd','residue','dtype','extra_file','duplicate_json','symlink'])

@pytest.mark.parametrize('cancel,fail_final_write',[(False,False),(True,False),(True,True)])
def test_collector_failure_cleanup_and_primary_cancellation(tmp_path,monkeypatch,cancel,fail_final_write):
 from types import SimpleNamespace
 import collect
 original_write=collect.process.write_json
 def fake_case(command,directory,**kwargs):
  record={'returncode':1,'passed':False,'error':'injected preprocessor failure','process_reaped':True,'public_key_store_removed':True,'owned_directory_handle_closed':True}
  original_write(directory/'process.json',record)
  if cancel:raise KeyboardInterrupt('primary cancellation')
  return record
 monkeypatch.setattr(collect.process,'run_case',fake_case)
 if fail_final_write:
  def failing_write(path,value):
   if path.name=='report.json' and 'elapsed_s' in value:raise OSError('secondary final write')
   return original_write(path,value)
  monkeypatch.setattr(collect.process,'write_json',failing_write)
 owned=[];original_mkdtemp=collect.tempfile.mkdtemp
 def own(*a,**k):
  result=original_mkdtemp(*a,dir=tmp_path,**k);owned.append(Path(result));return result
 monkeypatch.setattr(collect.tempfile,'mkdtemp',own)
 args=SimpleNamespace(output=tmp_path/'evidence',snapshot=tmp_path/'unused',preprocess_python=Path(sys.executable))
 if cancel:
  with pytest.raises(KeyboardInterrupt,match='primary cancellation') as captured:collect.run(args)
  if fail_final_write:assert any('secondary final write' in n for n in captured.value.__notes__)
 else:assert collect.run(args)==1
 assert owned and all(not p.exists() for p in owned)
 if not fail_final_write:
  r=json.loads((args.output/'report.json').read_text());assert r['passed'] is False and r['owned_component_directory_removed'] and r['all_processes_reaped']
