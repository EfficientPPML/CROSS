class Payload:
    dtype=np.dtype('uint32')
    def __init__(self,data=0.):self.data=np.asarray(data)
    def devices(self):return {'fake-cpu'}
    def astype(self,dtype):return self

class Value:
    modulus_dtype=np.uint32
    def __init__(self,scale,level,*,data=0.):
        self._ckks_scale=scale;self._ckks_nsd=1;self.level=level;self.moduli=Q[:2*(level+1)]
        self.polynomial=Payload(data);self.ntt_ctx=None;self.num_moduli=len(self.moduli)
        self.shape=(1,2,1,1,self.num_moduli)
    def _clone_with_payload(self,payload,**changes):
        result=copy.copy(self);result.polynomial=payload
        for key,value in changes.items():setattr(result,key,value)
        result.num_moduli=len(result.moduli);result.level=result.num_moduli//2-1;result.shape=(1,2,1,1,result.num_moduli)
        return result

def method(path,class_name,name):
    root=ast.parse(path.read_text());cls=next(n for n in root.body if isinstance(n,ast.ClassDef) and n.name==class_name)
    fn=next(n for n in cls.body if isinstance(n,ast.FunctionDef) and n.name==name)
    namespace={'np':np,'math':math,'Polynomial':Value,'finite_field':types.SimpleNamespace(check_binary_ct_operands=lambda *a,**k:None)}
    exec(compile(ast.Module(body=[fn],type_ignores=[]),str(path),'exec'),namespace)
    return namespace[name]

@pytest.mark.parametrize('level',(1,2,5,16,17,20,24,27,34,36,38,48,49))
@pytest.mark.parametrize('factor',(1.,.73,1.17))
def test_actual_compiled_metadata_matches_planner_operation_order(executor,level,factor):
    evaluate,checks=executor;source=DELTA*factor;divisor=math.prod(Q[2*level:2*level+2])
    result=evaluate([source],level,[.125]);expected=plan.weighted_sum_scale([source],DELTA,divisor)
    assert result._ckks_scale==expected['output_scale']
    assert result.level==level-1 and result._ckks_nsd==1
    assert len(checks)==2 # Actual plaintext operand and rescale operand checks executed.
    assert expected['terms'][0]['product_scale']==float(source)*float(expected['terms'][0]['plaintext_scale'])


def test_real_public_profile_has_one_ulp_counterexample_to_forced_delta(executor):
    evaluate,_=executor;result=evaluate([DELTA],38,[1.])
    assert result._ckks_scale==math.nextafter(DELTA,0.) and result._ckks_scale!=DELTA
    assert plan.weighted_sum_scale([DELTA],DELTA,plan.drop_divisor(Q,38))['output_scale']==result._ckks_scale


def test_zero_first_weight_still_owns_scale_and_reordering_changes_it(executor):
    evaluate,_=executor;level=38;divisor=plan.drop_divisor(Q,level)
    candidates=[DELTA*x for x in (.73,.81,1.,1.17,1.31)]
    pair=next((a,b) for a in candidates for b in candidates if
        plan.affine_scale(a,DELTA,divisor)['output_scale']!=plan.affine_scale(b,DELTA,divisor)['output_scale'])
    first=evaluate(pair,level,[0.,1.]);reversed_result=evaluate(pair[::-1],level,[1.,0.])
    assert np.array_equal(first.polynomial.data,reversed_result.polynomial.data)
    assert first._ckks_scale!=reversed_result._ckks_scale
    assert first._ckks_scale==plan.weighted_sum_scale(pair,DELTA,divisor)['output_scale']
    assert reversed_result._ckks_scale==plan.weighted_sum_scale(pair[::-1],DELTA,divisor)['output_scale']
