def test_six_sequential_no_deadline_exact_snapshot_and_cleanup(prepared,monkeypatch):
    repo,out,h=prepared;calls=[];original=h.run_case
    def track(*a,**kw):calls.append(kw.copy());return original(*a,**kw)
    monkeypatch.setattr(h,'run_case',track)
    assert m.evaluate(repo,out,h)==0
    report=json.loads((out/'report.json').read_text())
    assert report['passed'] and report['complete'] and report['source_files_unchanged']
    assert len(calls)==6 and all(c==dict(timeout=None,cleanup_grace_s=30.) for c in calls)
    assert len({c['pid'] for c in report['cases'].values()})==6
    assert all(c['process_reaped'] and c['public_key_store_removed'] and c['owned_directory_handle_closed'] for c in report['cases'].values())
    with tarfile.open(out/'sources.tar.gz') as tar:
        actual={v.name:hashlib.sha256(tar.extractfile(v).read()).hexdigest() for v in tar.getmembers()}
    assert actual==report['source_hashes_start']==report['source_hashes_end']
    assert len([n for n in actual if '/test/test' in n])==12

def test_publication_error_keeps_primary_cancellation(prepared,monkeypatch):
    repo,out,h=prepared;original=h.write_json
    def stop(*a,**k):raise KeyboardInterrupt('primary interruption')
    def write(path,obj):
        if path.name=='report.json' and 'elapsed_s' in obj:raise OSError('final publication failed')
        return original(path,obj)
    monkeypatch.setattr(h,'run_case',stop);monkeypatch.setattr(h,'write_json',write)
    with pytest.raises(KeyboardInterrupt,match='primary interruption') as caught:m.evaluate(repo,out,h)
    assert any('Final sweep evidence write failed' in n for n in caught.value.__notes__)

def test_outer_sigterm_waits_for_child_cleanup_and_preserves_failure(prepared,tmp_path):
    repo,out,h=prepared;started=tmp_path/'started';cleaned=tmp_path/'cleaned'
    child="import os,signal,time;from pathlib import Path\ndef stop(s,f):\n Path("+repr(str(cleaned))+").write_text('cleaned');raise SystemExit(0)\nsignal.signal(signal.SIGTERM,stop)\nPath("+repr(str(started))+").write_text(str(os.getpid()))\nwhile True:time.sleep(.05)\n"
    outer="import importlib.util;from pathlib import Path\ns=importlib.util.spec_from_file_location('p3six',"+repr(str(HERE/'check_p3_original_six.py'))+");m=importlib.util.module_from_spec(s);s.loader.exec_module(m)\nm.__doc__='CPU fake process cancellation only'\nm.command=lambda *a:["+repr(sys.executable)+",'-c',"+repr(child)+"]\nh=m.load_helpers(Path("+repr(str(repo))+"))\nwith m.termination_signals(h):m.evaluate(Path("+repr(str(repo))+"),Path("+repr(str(out))+"),h)\n"
    process=subprocess.Popen([sys.executable,'-c',outer],stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
    try:
        deadline=time.monotonic()+5
        while not started.exists() and time.monotonic()<deadline:time.sleep(.02)
        assert started.exists()
        process.send_signal(signal.SIGTERM)
        stdout,stderr=process.communicate(timeout=8)
        assert process.returncode==143 and cleaned.read_text()=='cleaned'
        report=json.loads((out/'report.json').read_text());record=report['cases']['test1.in']
        assert not report['passed'] and record['process_reaped'] and record['public_key_store_removed']
        assert not Path('/proc/'+started.read_text()).exists()
    finally:
        if process.poll() is None:process.kill();process.wait()
        if started.exists():
            try:os.kill(int(started.read_text()),signal.SIGKILL)
            except ProcessLookupError:pass
