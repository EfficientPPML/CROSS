"""Verify public preparation evidence without importing the crypto/model stack."""
from pathlib import Path
import hashlib,json,tarfile,xml.etree.ElementTree as ET
ROOT=Path(__file__).resolve().parent
sha=lambda raw:hashlib.sha256(raw).hexdigest()
def verify():
    report=json.loads((ROOT/'measured/report.json').read_text());summary=json.loads((ROOT/'summary.json').read_text())
    assert report['passed']is True and summary['actual_stage3_started']is False
    expected=report['source_hashes_start']
    assert len(expected)==76 and 'preflight_driver.py' in expected
    assert summary['proof_source_count_breakdown']=={'candidate_closure':75,'external_preflight_driver':1}
    assert expected==report['source_hashes_end']==report['snapshot_hashes_end']==report['archive_member_hashes']
    archive=ROOT/'measured/sources.tar.gz';assert sha(archive.read_bytes())==report['source_archive_sha256']==summary['source_archive_sha256']
    with tarfile.open(archive) as tar:
        members=tar.getmembers();assert len(members)==len({m.name for m in members})==76 and all(m.isfile() for m in members)
        raw={m.name:tar.extractfile(m).read() for m in members}
        assert {n:sha(b) for n,b in raw.items()}==expected
    origin=json.loads(raw['source_origin.json']);old={'runtime/'+n:h for n,h in origin['first_handler_runtime_source_hashes'].items()}|origin['additional_source']
    assert len(old)==59 and all(expected[n]==h for n,h in old.items())
    suites=list(ET.parse(ROOT/'measured/tests.xml').getroot().iter('testsuite'))
    assert sum(int(s.attrib.get('tests',0)) for s in suites)==115
    assert not any(int(s.attrib.get(n,0)) for s in suites for n in ('failures','errors','skipped'))
    probe=json.loads((ROOT/'measured/real_import_guard.json').read_text())
    assert probe['passed']is True and probe['denial_count']==probe['restored_count']==21
    assert len(set(probe['actual_entrypoints_denied']))==21
    assert not any(probe[n] for n in ('context_constructed','keys_generated','model_constructed','asset_payloads_read','encrypt_called','decrypt_called'))
    inventory=json.loads((ROOT/'archive_inventory.json').read_text())
    actual={p.relative_to(ROOT).as_posix():sha(p.read_bytes()) for p in ROOT.rglob('*') if p.is_file() and p.name!='archive_inventory.json' and '__pycache__' not in p.parts}
    assert actual==inventory
    print(json.dumps({'passed':True,'proof_sources':76,'candidate_sources':75,'mock_tests':115,'actual_entrypoint_denials':21,'actual_stage3':False}))
if __name__=='__main__':verify()
