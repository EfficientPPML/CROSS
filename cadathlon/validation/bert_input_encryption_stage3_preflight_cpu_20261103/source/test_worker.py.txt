def test_mock_handler_all_three_prerequisites_before_assets_and_encrypt(case):
    assert worker.run(**case.args)==0
    r=json.loads(case.output.read_text());assert r['passed'] and r['all_eight_ordered_payloads_validated']
    assert case.events[0]=='gates' and case.events.index('guard-enter')<case.events.index('encrypt')<case.events.index('readback')<case.events.index('guard-exit')
    assert r['source_hashes_start']==r['source_hashes_end']==case.expected
    assert r['remaining_artifacts']['upload']['present'] and r['remaining_artifacts']['private_requests']['present']


@pytest.mark.parametrize('gate',['key','model','input'])
def test_actual_gate_call_order_stops_before_next_prerequisite(monkeypatch,gate):
    events=[]
    def f(name,result):
        def call(*a,**kw):
            events.append(name)
            if name==gate:raise ValueError(name+' prerequisite')
            return result
        return call
    monkeypatch.setattr(worker.key_gate,'authenticate',f('key',{}));monkeypatch.setattr(worker.model_gate,'authenticate',f('model',{}))
    monkeypatch.setitem(sys.modules,'input_gate',NS(authenticate=f('input',{})))
    with pytest.raises(ValueError,match=gate):worker.authenticate(key_evidence=0,key_report_sha256=0,key_workspace=0,model_evidence=0,model_report_sha256=0,model_workspace=0,fixture_evidence=0,fixture_report_sha256=0,profile_bytes=b'',plan_bytes=b'')
    assert events==['key','model','input'][:['key','model','input'].index(gate)+1]


def test_failed_gate_reads_no_assets_or_input_payload(case,monkeypatch):
    monkeypatch.setattr(worker,'authenticate',lambda **kw:(_ for _ in ()).throw(ValueError('gate failed')))
    with pytest.raises(ValueError,match='gate failed'):worker.run(**case.args)
    r=json.loads(case.output.read_text());assert not r['passed'] and not r['actual_client_encode_started']
    assert case.events==[] and not (case.io/'upload').exists()

@pytest.mark.parametrize('kind',['handler','readback','restore'])
def test_started_failure_retains_remaining_upload_and_private_pair(case,kind):
    error=KeyboardInterrupt(kind);case.failure[kind]=error
    with pytest.raises(KeyboardInterrupt) as caught:worker.run(**case.args)
    assert caught.value is error
    r=json.loads(case.output.read_text());assert not r['passed'] and r['actual_client_encode_started']
    assert (case.io/'upload').is_dir() and (case.keys/'client_private/requests.json').is_file()
    assert 'guard-exit' in case.events

    assert worker.run(**case.args)==1
    r=json.loads(case.output.read_text());assert not r['passed'] and not r['source_files_unchanged']


def test_initial_publication_failure_recovers_fresh_failure_report(case,monkeypatch):
    original=worker.write;calls=[]
    def write(path,value):
        calls.append(1)
        if len(calls)==1:raise OSError('initial publication')
        return original(path,value)
    monkeypatch.setattr(worker,'write',write)
    with pytest.raises(OSError,match='initial publication'):worker.run(**case.args)
    assert not json.loads(case.output.read_text())['passed'] and not case.events


def test_final_publication_preserves_primary_cancellation(case,monkeypatch):
    original=worker.write;error=KeyboardInterrupt('primary');case.failure['readback']=error
    def write(path,value):
        if 'source_hashes_end' in value:raise OSError('terminal publication')
        original(path,value)
    monkeypatch.setattr(worker,'write',write)
    with pytest.raises(KeyboardInterrupt) as caught:worker.run(**case.args)
    assert caught.value is error and any('terminal publication' in n for n in error.__notes__)
