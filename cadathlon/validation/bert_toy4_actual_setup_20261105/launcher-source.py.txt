def read(p,h=None):
 p=Path(p);assert stat.S_ISREG(p.lstat().st_mode)

 except ProcessLookupError:return

assert len(manifest)==260 and prep['source_manifest_sha256']=='ec2bb07eb7689b2b3ec6851b0c62c851200473afe716af27331d0655d3a0239e'

  assert m.isfile() and not m.issym() and not m.islnk()
  assert sha(t.extractfile(m).read())==manifest[m.name]

assert sha(python.read_bytes())==prep['interpreter']['sha256']
assert manifest['collect.py']==prep['initial_script_sha256']

vfio=['/dev/vfio/vfio']+['/dev/vfio/'+str(i) for i in range(8)]+['/dev/vfio/devices/vfio'+str(i) for i in range(8)]

parent=None

  command=command,cwd=str(SRC),environment={name:env.get(name) for name in (*unset,'PYTHONDONTWRITEBYTECODE','PYTHONPATH','OMP_NUM_THREADS','OPENBLAS_NUM_THREADS','MKL_NUM_THREADS','NUMEXPR_NUM_THREADS','JAX_PLATFORMS','CROSS_ROOT')},
  source_manifest_sha256=prep['source_manifest_sha256'],source_archive_sha256=prep['source_archive_sha256'],preparation_sha256=sha(prep_raw),
  activation_review=str(a.activation_review),activation_review_sha256=sha(review),P3_terminal_admission_sha256=sha(prior_raw),P3_original_terminal_chunk=prior['terminal_tool_chunk'],
  owned_workspace_parent=str(parent),owned_workspace_parent_identity=dict(device=s.st_dev,inode=s.st_ino,uid=s.st_uid,gid=s.st_gid,mode=stat.S_IMODE(s.st_mode)),
  workspace_retention='Collector owns fresh nested workspace; retain it and this parent across separately reviewed roles. No cleanup by guessed names or uncertain identity.',

  checked_free_devices=vfio,device_fuser_returncode=1,healthy_timeout=None,server_or_validator_automatically_launched=False,

except BaseException as primary:

  primary.add_note('Returned parent path requiring retention: '+repr(failure['acquired_parent_path']))
 cleanup=dict(parent=parent_path,removed=False,retained=True,reason='identity unavailable; no guessed cleanup')
 if parent is not None and owned_identity is not None:
  try:
   current=parent.lstat()
   identity=(current.st_dev,current.st_ino,current.st_uid,current.st_gid,stat.S_IMODE(current.st_mode))
   if not stat.S_ISDIR(current.st_mode) or identity!=owned_identity:
    raise RuntimeError('launch parent identity changed; left untouched')

    raise RuntimeError('launch parent identity changed before removal; retained')
