HERE = Path(__file__).resolve().parent

N, SLOTS, F, TILES = 131072, 65536, 4096, 8
ring_file = ROOT/'cadathlon/validation/bert_factored_routing_secure_cpu_20261103/report.json'
linear_file = ROOT/'cadathlon/validation/streaming_linear_qkv_cpu_20261103/measured/public_manifest.json'
ring = json.loads(ring_file.read_text())['ring']
measured = json.loads(linear_file.read_text())
linear = {r if r < 4096 else r-8192+SLOTS for r in measured['rotations']}
active = {d % SLOTS for d in range(-767, 768)}
babies = {d % 64 for d in active} - {0}
giants = {d // 64 * 64 for d in active} - {0}
assert linear == babies | giants and len(linear) == 86
head = {1 << bit for bit in range(6)}
token = {F*offset for offset in range(1, 16)}
forward = ({63*(1 << bit) for bit in range(4)}
           | {(-16*(1 << bit)) % SLOTS for bit in range(4)}
           | {(-256*tile) % SLOTS for tile in range(1, 8)})
direct = linear | head | token | forward
representative = linear | head | {15*F} | ({63*(1 << bit) for bit in range(4)}
           | {(-16*(1 << bit)) % SLOTS for bit in range(4)} | {(-256*7) % SLOTS})
assert len(direct) == 110 and len(representative) == 93

# An explicit composition uses existing BSGS keys for every forward-map shift.
# Only four positive row-power keys are added. Summed shifts prove index parity;
# this does not establish accumulated key-switch noise.
parts = {63: [63], 126: [64, 62], 252: [192, 60], 504: [448, 56],
    -16: [-64, 48], -32: [-64, 32], -64: [-64], -128: [-128],
    0: [], -256: [-256], -512: [-512], -768: [-768],
    -1024: [-512, -512], -1280: [-768, -512],
    -1536: [-768, -768], -1792: [-768, -768, -256]}
for shift, sequence in parts.items():
    assert sum(sequence) % SLOTS == shift % SLOTS
    assert all(part % SLOTS in linear for part in sequence)
row_power = {F*(1 << bit) for bit in range(4)}
bounded = linear | row_power
for offset in range(16):
    assert sum(F*(1 << bit) for bit in range(4) if offset & (1 << bit)) == offset*F
assert len(bounded) == 90
maps = TILES*TILES*16
forward_rotations = TILES*sum(8 + int(tile != 0) for tile in range(8) for _ in range(16))
composed_forward = TILES*sum(sum(len(parts[x]) for x in (63,126,252,504,-16,-32,-64,-128,-256*tile))
                            for tile in range(8) for _ in range(16))
row_rotations = TILES*15
composed_rows = TILES*sum(i.bit_count() for i in range(16))
linear_rotations = 16*len(linear)
head_rotations = 6*maps
key_bytes = 2*4*(38+10)*N*4
ct = lambda level: 2*N*(2*(level+1))*4
